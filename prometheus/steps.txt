# Step 1: Download and extract Prometheus binary
wget https://github.com/prometheus/prometheus/releases/download/v2.54.0/prometheus-2.54.0.linux-amd64.tar.gz
tar xvfz prometheus-2.54.0.linux-amd64.tar.gz
cd prometheus-2.54.0.linux-amd64

# Step 2: Configure scrape targets in prometheus.yml
cat << 'EOF' > prometheus.yml
global:
  scrape_interval: 15s
  evaluation_interval: 15s

scrape_configs:
  - job_name: "prometheus"
    static_configs:
      - targets: ["localhost:9090"]

  - job_name: "mumbai_worker"
    static_configs:
      - targets: ["13.127.221.88:9100"]

  - job_name: "virginia_worker"
    static_configs:
      - targets: ["107.23.241.198:9100"]
EOF

# Step 3: Verify configuration and run Prometheus server
cat prometheus.yml
cd ~/prometheus-2.54.0.linux-amd64
./prometheus --config.file=prometheus.yml
